test('CRS-01-SC-004', async ({ page }) => {

  await login(page);
  // PROOF: Publish two distinct immutable versions and verify history shows both titles
  await page.getByRole('button', { name: 'Courses' }).click();
  await page.getByRole('button', { name: 'Create course' }).click();

  // First version
  await page.getByLabel('Course title').fill('Version One');
  await page.getByLabel('Stage 1 title').fill('Stage A');
  await page.getByLabel('Stage 1 objective 1').fill('Objective A1');
  await page.getByRole('button', { name: 'Save draft' }).click();
  await page.getByRole('button', { name: 'Publish' }).click();
  await page.getByRole('button', { name: 'Confirm publish' }).click();

  // Verify version 1 appears
  await expect(page.getByText('Version 1')).toBeVisible();
  await expect(page.getByText('Version One')).toBeVisible();

  // Edit title and publish second version
  await page.getByRole('button', { name: 'Edit' }).click();
  await page.getByLabel('Course title').fill('Version Two');
  await page.getByRole('button', { name: 'Publish' }).click();
  await page.getByRole('button', { name: 'Confirm publish' }).click();

  // Verify history shows both immutable versions with distinct titles
  await page.getByRole('button', { name: 'History' }).click();
  await expect(page.getByText('Version 1')).toBeVisible();
  await expect(page.getByText('Version One')).toBeVisible();
  await expect(page.getByText('Version 2')).toBeVisible();
  await expect(page.getByText('Version Two')).toBeVisible();

  // Verify backend recorded two publish calls
  const calls = await page.evaluate(() => window.__coachFixture.calls);
  const publishCalls = calls.filter(c => c === 'publishCourse');
  expect(publishCalls.length).toBe(2);
});
